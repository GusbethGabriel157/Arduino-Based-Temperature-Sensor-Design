long readVcc() {

  delay(2); // timp de stabilizare

  while (bit_is_set(ADCSRA, ADSC));

  uint16_t result = ADC;

  // 1.1V * 1023 * 1000 (pentru mV)
  long vcc = 1125300L / result;
